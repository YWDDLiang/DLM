"""Resumable full S.U.N. evaluation.

This wraps eval_sun.py's novelty/uniqueness and MP hull code, but stores
CHGNet relaxation results incrementally so long full runs can be resumed.
"""
import argparse
import hashlib
import json
import os
import time
from pathlib import Path

import numpy as np
from monty.json import MontyDecoder
from pymatgen.analysis.structure_matcher import StructureMatcher
from pymatgen.core import Composition
from tqdm import tqdm

from eval_sun import (
    compute_e_hull_batch,
    compute_novelty,
    compute_uniqueness,
    load_generated_structures,
    load_training_index,
)


def decode_composition(payload):
    if isinstance(payload, dict) and "@module" in payload:
        return MontyDecoder().process_decoded(payload)
    return Composition.from_dict(payload)


def structure_cache_key(structure, ndigits=6):
    payload = {
        "lattice": np.round(structure.lattice.matrix, ndigits).tolist(),
        "sites": [
            {
                "species": str(site.specie),
                "frac_coords": np.round(site.frac_coords, ndigits).tolist(),
            }
            for site in structure
        ],
    }
    encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
    return hashlib.sha256(encoded.encode("utf-8")).hexdigest()


def load_relax_results(path):
    results = {}
    if not path.exists():
        return results
    with path.open("r", encoding="utf-8") as f:
        for line in f:
            if not line.strip():
                continue
            item = json.loads(line)
            comp = decode_composition(item["composition"])
            results[int(item["local_index"])] = (item["energy_per_atom"], comp)
    return results


def load_global_relax_cache(path):
    results = {}
    if path is None or not path.exists():
        return results
    with path.open("r", encoding="utf-8") as f:
        for line in f:
            if not line.strip():
                continue
            item = json.loads(line)
            comp = decode_composition(item["composition"])
            results[item["structure_key"]] = (item["energy_per_atom"], comp)
    return results


def append_relax_result(path, local_index, energy_per_atom, composition):
    path.parent.mkdir(parents=True, exist_ok=True)
    item = {
        "local_index": int(local_index),
        "energy_per_atom": energy_per_atom,
        "composition": composition.as_dict(),
    }
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(item, separators=(",", ":")) + "\n")
        f.flush()
        os.fsync(f.fileno())


def append_global_relax_result(path, structure_key, energy_per_atom, composition):
    if path is None:
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    item = {
        "structure_key": structure_key,
        "energy_per_atom": energy_per_atom,
        "composition": composition.as_dict(),
    }
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(item, separators=(",", ":")) + "\n")
        f.flush()
        os.fsync(f.fileno())


def append_global_relax_results(path, items):
    if path is None or not items:
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("a", encoding="utf-8") as f:
        for structure_key, energy_per_atom, composition in items:
            item = {
                "structure_key": structure_key,
                "energy_per_atom": energy_per_atom,
                "composition": composition.as_dict(),
            }
            f.write(json.dumps(item, separators=(",", ":")) + "\n")
        f.flush()
        os.fsync(f.fileno())


def relax_missing(structures, relax_path, device, global_relax_path=None):
    from chgnet.model import StructOptimizer

    existing = load_relax_results(relax_path)
    global_existing = load_global_relax_cache(global_relax_path)
    optimizer = StructOptimizer(use_device=device)
    failed = 0
    global_hits = 0

    for i, structure in enumerate(tqdm(structures, desc="Resumable CHGNet relaxation")):
        if i in existing:
            continue
        structure_key = structure_cache_key(structure)
        if structure_key in global_existing:
            energy_per_atom, comp = global_existing[structure_key]
            append_relax_result(relax_path, i, energy_per_atom, comp)
            global_hits += 1
            continue
        try:
            result = optimizer.relax(structure, verbose=False)
            relaxed = result["final_structure"]
            trajectory = result.get("trajectory")
            if trajectory is None or not getattr(trajectory, "energies", None):
                raise ValueError("CHGNet relaxation did not return energies")
            energy_per_atom = float(trajectory.energies[-1]) / relaxed.num_sites
            comp = relaxed.composition
        except Exception:
            failed += 1
            energy_per_atom = None
            comp = structure.composition
        append_relax_result(relax_path, i, energy_per_atom, comp)
        append_global_relax_result(global_relax_path, structure_key, energy_per_atom, comp)
        global_existing[structure_key] = (energy_per_atom, comp)

    all_results = load_relax_results(relax_path)
    ordered = [all_results[i] for i in range(len(structures)) if i in all_results]
    print(f"Relaxation records: {len(ordered)}/{len(structures)}")
    print(f"Global relaxation cache hits: {global_hits}")
    print(f"New relaxation failures in this run: {failed}")
    return ordered


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--gen_file", required=True)
    parser.add_argument("--train_csv", required=True)
    parser.add_argument("--output_dir", required=True)
    parser.add_argument("--stable_threshold", type=float, default=0.0)
    parser.add_argument("--mp_api_key_env", default=None)
    parser.add_argument("--mp_cache_path", default="data/mp_hull_entries_cache_resumable.jsonl")
    parser.add_argument("--global_relax_cache_path", default="data/chgnet_relax_cache_global.jsonl")
    parser.add_argument("--no_cuda", action="store_true")
    args = parser.parse_args()

    device = "cpu" if args.no_cuda else "cuda"
    out_dir = Path(args.output_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    relax_path = out_dir / "relax_results.jsonl"
    summary_path = out_dir / "RESULTS_SUMMARY.md"

    matcher = StructureMatcher(ltol=0.2, stol=0.3, angle_tol=5)
    gen_structures, n_generated_total = load_generated_structures(args.gen_file)
    train_structures, train_formula_idx = load_training_index(args.train_csv)

    novel_mask = compute_novelty(gen_structures, train_structures, train_formula_idx, matcher)
    n_novel = int(novel_mask.sum())
    eq_class, n_unique = compute_uniqueness(gen_structures, matcher)

    seen_classes = set()
    novel_unique_mask = np.zeros(len(gen_structures), dtype=bool)
    for i in range(len(gen_structures)):
        cls = eq_class[i]
        if cls in seen_classes:
            continue
        seen_classes.add(cls)
        if novel_mask[i]:
            novel_unique_mask[i] = True

    n_novel_unique = int(novel_unique_mask.sum())
    novel_unique_structures = [s for s, m in zip(gen_structures, novel_unique_mask) if m]
    print(f"Novel + Unique: {n_novel_unique}/{len(gen_structures)}")

    t0 = time.time()
    global_relax_path = Path(args.global_relax_cache_path) if args.global_relax_cache_path else None
    energies_comps = relax_missing(novel_unique_structures, relax_path, device, global_relax_path)
    print(f"Relaxation stage wall time: {time.time() - t0:.0f}s")
    if len(energies_comps) < len(novel_unique_structures):
        raise RuntimeError("Relaxation records incomplete; rerun to resume.")

    mp_api_key = os.environ.get(args.mp_api_key_env) if args.mp_api_key_env else None
    t0 = time.time()
    e_hull_results = compute_e_hull_batch(
        energies_comps,
        args.train_csv,
        mp_api_key=mp_api_key,
        mp_cache_path=args.mp_cache_path,
    )
    print(f"E_hull computation: {time.time() - t0:.0f}s")

    n_stable = 0
    n_evaluated = 0
    n_unknown = 0
    for energy, comp in energies_comps:
        if energy is None:
            n_unknown += 1
            continue
        e_hull = e_hull_results.get((energy, comp.reduced_formula))
        if e_hull is None:
            n_unknown += 1
        else:
            n_evaluated += 1
            if e_hull <= args.stable_threshold:
                n_stable += 1

    stable_pct = 100 * n_stable / n_evaluated if n_evaluated else 0
    sun_pct = 100 * n_stable / len(gen_structures)
    estimated_sun_pct = 100 * (n_stable / n_evaluated) * (n_novel_unique / len(gen_structures)) if n_evaluated else 0

    lines = [
        "# Resumable Full S.U.N. Summary",
        f"generated: {args.gen_file}",
        f"Total generated: {n_generated_total}",
        f"Reconstructed: {len(gen_structures)}",
        f"Novel: {n_novel}/{len(gen_structures)} ({100*n_novel/len(gen_structures):.2f}%)",
        f"Unique: {n_unique}/{len(gen_structures)} ({100*n_unique/len(gen_structures):.2f}%)",
        f"Novel + Unique: {n_novel_unique}/{len(gen_structures)} ({100*n_novel_unique/len(gen_structures):.2f}%)",
        f"E_hull evaluated: {n_evaluated}/{len(energies_comps)}",
        f"E_hull unknown: {n_unknown}/{len(energies_comps)}",
        f"Stable: {n_stable}/{n_evaluated} evaluated ({stable_pct:.2f}%)",
        f"Full S.U.N. lower-bound: {sun_pct:.2f}%",
        f"Coverage-adjusted S.U.N. estimate: {estimated_sun_pct:.2f}%",
    ]
    summary_path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print("\n".join(lines))
    print(f"Saved summary to {summary_path}")


if __name__ == "__main__":
    main()
