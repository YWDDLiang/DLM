"""
S.U.N. (Stable, Unique, Novel) evaluation for crystal generation.

Based on PLaID++ methodology:
- Stable: relaxation/prediction -> E_hull <= threshold
- Unique: StructureMatcher deduplication within generated set
- Novel: StructureMatcher comparison against training set

Optimization: structures are indexed by reduced formula to avoid
comparing crystals with different compositions.

Usage:
    python eval_sun.py --gen_file results/llama3_sample_5epoch_mp_20_10000.pt \
                       --train_csv data/mp_20/train.csv --skip_stability

Notes:
    --skip_stability reports Novel+Unique only. It is SUN-lite, not full S.U.N.
    Full S.U.N. requires generated energies that are compatible with the
    reference hull. The MP-20 CSV used by this project does not contain total
    energies, so the script refuses to report a misleading full S.U.N. number
    unless a compatible hull construction is available.
"""

import argparse
import hashlib
import json
import os
import pickle
import time
import warnings
import numpy as np
import pandas as pd
import torch
from collections import defaultdict
from pathlib import Path
from tqdm import tqdm
from monty.json import MontyDecoder

from pymatgen.core import Structure, Lattice
from pymatgen.analysis.structure_matcher import StructureMatcher
from pymatgen.entries.computed_entries import ComputedEntry

warnings.filterwarnings("ignore")


def load_generated_structures(pt_file):
    """Reconstruct pymatgen Structures from .pt file."""
    data = torch.load(pt_file, map_location="cpu")
    frac_coords = data["frac_coords"][0]
    num_atoms = data["num_atoms"][0]
    atom_types = data["atom_types"][0]
    lengths = data["lengths"][0]
    angles = data["angles"][0]

    n_total = len(num_atoms)
    structures = []

    offset = 0
    for i in tqdm(range(n_total), desc="Reconstructing"):
        n_atoms = int(num_atoms[i])
        try:
            coords = frac_coords[offset : offset + n_atoms].numpy()
            types = atom_types[offset : offset + n_atoms].numpy()
            length = lengths[i].numpy().tolist()
            angle = angles[i].numpy().tolist()
            lattice = Lattice.from_parameters(*length, *angle)
            structure = Structure(
                lattice=lattice, species=types, coords=coords,
                coords_are_cartesian=False,
            )
            if structure.volume >= 0.1:
                structures.append(structure)
        except Exception:
            pass
        offset += n_atoms

    print(f"Reconstructed {len(structures)}/{n_total} structures")
    return structures, n_total


def index_by_formula(structures):
    """Build dict: reduced_formula -> list of structure indices."""
    idx = defaultdict(list)
    for i, s in enumerate(structures):
        idx[s.composition.reduced_formula].append(i)
    return idx


def load_training_index(csv_path):
    """Load training structures, indexed by reduced formula."""
    csv_path = Path(csv_path)
    cache_path = csv_path.parent / ".cache" / f"{csv_path.name}.{_file_fingerprint(csv_path)}.training_index.pkl"
    if cache_path.exists():
        try:
            with cache_path.open("rb") as f:
                structures, formula_items = pickle.load(f)
            formula_idx = defaultdict(list)
            formula_idx.update({k: list(v) for k, v in formula_items})
            print(
                f"Loaded {len(structures)} cached training structures, "
                f"{len(formula_idx)} unique formulas"
            )
            return structures, formula_idx
        except Exception as exc:
            print(f"Warning: ignored unreadable training cache {cache_path}: {exc}")

    df = pd.read_csv(csv_path)
    formula_idx = defaultdict(list)
    structures = []

    for _, row in tqdm(df.iterrows(), total=len(df), desc="Loading training"):
        try:
            s = Structure.from_str(row["cif"], fmt="cif")
            formula_idx[s.composition.reduced_formula].append(len(structures))
            structures.append(s)
        except Exception:
            pass

    print(f"Loaded {len(structures)} training structures, "
          f"{len(formula_idx)} unique formulas")
    try:
        cache_path.parent.mkdir(parents=True, exist_ok=True)
        with cache_path.open("wb") as f:
            pickle.dump((structures, sorted(formula_idx.items())), f)
    except Exception as exc:
        print(f"Warning: could not write training cache {cache_path}: {exc}")
    return structures, formula_idx


def _file_fingerprint(path):
    stat = path.stat()
    payload = f"{path.resolve()}:{stat.st_size}:{stat.st_mtime_ns}"
    return hashlib.sha1(payload.encode("utf-8")).hexdigest()[:16]


def compute_novelty(gen_structures, train_structures, train_formula_idx, matcher):
    """Novelty check using composition-indexed search."""
    gen_formula_idx = index_by_formula(gen_structures)
    novel_mask = np.ones(len(gen_structures), dtype=bool)

    total_comparisons = 0
    for formula, gen_indices in tqdm(gen_formula_idx.items(),
                                     desc="Novelty check"):
        if formula not in train_formula_idx:
            continue  # all structures with this formula are novel

        train_indices = train_formula_idx[formula]
        for gi in gen_indices:
            gen = gen_structures[gi]
            for ti in train_indices:
                total_comparisons += 1
                if matcher.fit(gen, train_structures[ti]):
                    novel_mask[gi] = False
                    break

    print(f"Novelty comparisons: {total_comparisons}")
    return novel_mask


def compute_uniqueness(structures, matcher):
    """Deduplicate within generated set, using formula-indexed approach."""
    formula_idx = index_by_formula(structures)
    eq_class = np.full(len(structures), -1, dtype=int)
    class_id = 0

    for formula, indices in tqdm(formula_idx.items(), desc="Uniqueness"):
        for i in indices:
            matched = False
            # Only check previously seen structures with same formula
            for j in indices:
                if j >= i:
                    break
                if eq_class[j] >= 0 and matcher.fit(structures[i], structures[j]):
                    eq_class[i] = eq_class[j]
                    matched = True
                    break
            if not matched:
                eq_class[i] = class_id
                class_id += 1

    n_unique = len(set(eq_class))
    print(f"Unique: {n_unique}/{len(structures)} ({100*n_unique/len(structures):.2f}%)")
    return eq_class, n_unique


def compute_stability_chgnet(structures, device="cuda"):
    """CHGNet relaxation -> (energy_per_atom, composition)."""
    from chgnet.model import StructOptimizer
    optimizer = StructOptimizer(use_device=device)

    results = []
    failed = 0
    for s in tqdm(structures, desc="CHGNet relaxation"):
        try:
            result = optimizer.relax(s, verbose=False)
            relaxed = result["final_structure"]
            trajectory = result.get("trajectory")
            if trajectory is None or not getattr(trajectory, "energies", None):
                raise ValueError("CHGNet relaxation did not return energies")
            energy_per_atom = float(trajectory.energies[-1]) / relaxed.num_sites
            results.append((energy_per_atom, relaxed.composition))
        except Exception:
            failed += 1
            results.append((None, s.composition))
    print(f"CHGNet relaxation failures: {failed}/{len(structures)}")
    return results


def _load_mp_entry_cache(cache_path):
    if cache_path is None:
        return {}
    path = Path(cache_path)
    if not path.exists():
        return {}
    if path.suffix == ".jsonl":
        cache = {}
        decoder = MontyDecoder()
        with path.open("r", encoding="utf-8") as f:
            for line_no, line in enumerate(f, start=1):
                line = line.strip()
                if not line:
                    continue
                try:
                    record = json.loads(line)
                    entries = _decode_mp_cache_entries(record.get("entries"), decoder)
                    cache[record["chemsys"]] = entries
                except Exception as exc:
                    print(f"Warning: skipped bad JSONL cache line {line_no}: {exc}")
        return cache
    try:
        with path.open("rb") as f:
            cache = pickle.load(f)
        decoder = MontyDecoder()
        return {
            chemsys: _decode_mp_cache_entries(entries, decoder)
            for chemsys, entries in cache.items()
        }
    except (EOFError, pickle.UnpicklingError):
        corrupt_path = path.with_suffix(path.suffix + ".corrupt")
        path.replace(corrupt_path)
        print(f"Warning: MP entry cache was unreadable; moved to {corrupt_path}")
        return {}


def _save_mp_entry_cache(cache_path, cache):
    if cache_path is None:
        return
    path = Path(cache_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    if path.suffix == ".jsonl":
        tmp_path = path.with_suffix(path.suffix + ".tmp")
        with tmp_path.open("w", encoding="utf-8") as f:
            for chemsys, entries in sorted(cache.items()):
                record = {"chemsys": chemsys, "entries": _slim_mp_entries(entries)}
                f.write(json.dumps(record, separators=(",", ":")) + "\n")
        tmp_path.replace(path)
        return
    with path.open("wb") as f:
        pickle.dump({k: _slim_mp_entries(v) for k, v in cache.items()}, f)


def _append_mp_entry_cache(cache_path, chemsys, entries):
    if cache_path is None:
        return
    path = Path(cache_path)
    if path.suffix != ".jsonl":
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    record = {"chemsys": chemsys, "entries": _slim_mp_entries(entries)}
    with path.open("a", encoding="utf-8") as f:
        f.write(json.dumps(record, separators=(",", ":")) + "\n")


def _slim_mp_entries(entries):
    """Keep only fields needed for PhaseDiagram construction."""
    if entries is None:
        return None
    slim = []
    for entry in entries:
        if isinstance(entry, dict) and set(entry).issubset({"entry_id", "composition", "energy"}):
            slim.append(
                {
                    "entry_id": entry.get("entry_id"),
                    "composition": entry["composition"],
                    "energy": float(entry["energy"]),
                }
            )
        else:
            slim.append(
                {
                    "entry_id": getattr(entry, "entry_id", None),
                    "composition": entry.composition.as_dict(),
                    "energy": float(entry.energy),
                }
            )
    return slim


def _decode_mp_cache_entries(entries, decoder):
    if entries is None:
        return None
    if not entries:
        return []

    first = entries[0]
    if isinstance(first, dict) and set(first).issubset({"entry_id", "composition", "energy"}):
        return [
            ComputedEntry(
                entry["composition"],
                float(entry["energy"]),
                entry_id=entry.get("entry_id"),
            )
            for entry in entries
        ]

    return decoder.process_decoded(entries)


def _get_mp_entries_for_chemsys(
    mpr, elements, cache, cache_path=None, retries=5, sleep_s=10
):
    chemsys = "-".join(sorted(elements))
    if chemsys in cache:
        return cache[chemsys]
    last_error = None
    for attempt in range(1, retries + 1):
        try:
            try:
                entries = mpr.get_entries_in_chemsys(elements, compatible_only=True)
            except TypeError:
                entries = mpr.get_entries_in_chemsys(elements)
            entries = _decode_mp_cache_entries(_slim_mp_entries(entries), MontyDecoder())
            cache[chemsys] = entries
            _append_mp_entry_cache(cache_path, chemsys, entries)
            return entries
        except Exception as exc:
            last_error = exc
            print(f"MP query failed for {chemsys} ({attempt}/{retries}): {exc}")
            if attempt < retries:
                time.sleep(sleep_s * attempt)
    print(f"MP query abandoned for {chemsys}: {last_error}")
    return None


def compute_e_hull_batch(
    energies_comps,
    train_csv,
    mp_api_key=None,
    mp_cache_path="data/mp_hull_entries_cache.jsonl",
):
    """Compute e_hull for generated structures.

    This function intentionally refuses to mix incompatible energies. MP-20
    train.csv contains formation_energy_per_atom and e_above_hull, but not DFT
    total energies. CHGNet relaxed energies cannot be safely combined with that
    CSV to build a rigorous phase diagram.
    """
    from pymatgen.analysis.phase_diagram import PhaseDiagram, PDEntry

    if mp_api_key or mp_cache_path:
        cache = _load_mp_entry_cache(mp_cache_path)
        results = {}
        grouped = defaultdict(list)
        for energy, comp in energies_comps:
            if energy is None:
                continue
            elements = tuple(sorted(el.symbol for el in comp.elements))
            grouped[elements].append((energy, comp))

        mpr_context = None
        if mp_api_key:
            from pymatgen.ext.matproj import MPRester

            mpr_context = MPRester(mp_api_key)

        try:
            mpr = mpr_context.__enter__() if mpr_context else None
            for elements, items in tqdm(grouped.items(), desc="MP hull queries"):
                chemsys = "-".join(sorted(elements))
                if chemsys in cache:
                    entries = cache[chemsys]
                elif mpr is not None:
                    entries = _get_mp_entries_for_chemsys(
                        mpr, list(elements), cache, mp_cache_path
                    )
                else:
                    entries = None

                if not entries:
                    for energy, comp in items:
                        results[(energy, comp.reduced_formula)] = None
                    continue

                try:
                    phase_diagram = PhaseDiagram(entries)
                except Exception:
                    phase_diagram = None

                for energy, comp in items:
                    entry = PDEntry(comp, energy * comp.num_atoms)
                    try:
                        if phase_diagram is None:
                            raise ValueError("Could not build reference phase diagram")
                        _, e_hull = phase_diagram.get_decomp_and_e_above_hull(
                            entry,
                            allow_negative=True,
                        )
                        results[(energy, comp.reduced_formula)] = max(float(e_hull), 0.0)
                    except Exception:
                        results[(energy, comp.reduced_formula)] = None
        finally:
            if mpr_context is not None:
                mpr_context.__exit__(None, None, None)

        if mp_cache_path and Path(mp_cache_path).suffix != ".jsonl":
            _save_mp_entry_cache(mp_cache_path, cache)
        return results

    df = pd.read_csv(train_csv)
    if "energy" not in df.columns and "total_energy" not in df.columns:
        raise ValueError(
            "Cannot compute full S.U.N. stability from this train_csv: "
            "no compatible total-energy column was found. "
            "Use --skip_stability for SUN-lite, or provide a reference hull "
            "with energies compatible with the generated energies. "
            "If you have a Materials Project key, set MP_API_KEY and pass "
            "--mp_api_key_env MP_API_KEY."
        )

    energy_col = "energy" if "energy" in df.columns else "total_energy"
    train_entries = []
    for _, row in tqdm(df.iterrows(), total=len(df), desc="Building convex hull"):
        try:
            s = Structure.from_str(row["cif"], fmt="cif")
            train_entries.append(PDEntry(s.composition, float(row[energy_col])))
        except Exception:
            pass

    if not train_entries:
        raise ValueError("No valid reference entries were built for the hull.")

    # Add MP reference entries for hull construction
    # Group by composition to build per-composition phase diagrams
    from collections import Counter
    comp_groups = defaultdict(list)
    for energy, comp in energies_comps:
        if energy is not None:
            comp_groups[comp.reduced_formula].append((energy, comp))

    results = {}
    # For efficiency, build one big PhaseDiagram with all train entries
    # and query each generated entry
    all_entries = list(train_entries)
    phase_diagram = PhaseDiagram(all_entries)

    for formula, items in tqdm(comp_groups.items(), desc="Computing e_hull"):
        for energy, comp in items:
            entry = PDEntry(comp, energy * comp.num_atoms)
            try:
                _, e_hull = phase_diagram.get_decomp_and_e_above_hull(
                    entry,
                    allow_negative=True,
                )
                results[(energy, formula)] = max(float(e_hull), 0.0)
            except Exception:
                results[(energy, formula)] = None

    return results


def main():
    parser = argparse.ArgumentParser(description="S.U.N. evaluation")
    parser.add_argument("--gen_file", required=True)
    parser.add_argument("--train_csv", required=True)
    parser.add_argument("--skip_stability", action="store_true")
    parser.add_argument("--stability_n", type=int, default=None)
    parser.add_argument("--stable_threshold", type=float, default=0.0,
                        help="E_hull threshold in eV/atom for full S.U.N.")
    parser.add_argument("--mp_api_key_env", default=None,
                        help="Environment variable containing Materials Project API key.")
    parser.add_argument("--mp_cache_path", default="data/mp_hull_entries_cache.jsonl",
                        help="JSONL cache for Materials Project hull entries.")
    parser.add_argument("--no_cuda", action="store_true")
    args = parser.parse_args()

    device = "cpu" if args.no_cuda else "cuda"
    matcher = StructureMatcher(ltol=0.2, stol=0.3, angle_tol=5)

    # ---- Load data ----
    print("=" * 60)
    print("S.U.N. Evaluation (PLaID++ methodology)")
    print("=" * 60)

    t0 = time.time()
    gen_structures, n_generated_total = load_generated_structures(args.gen_file)
    train_structures, train_formula_idx = load_training_index(args.train_csv)
    print(f"Load time: {time.time() - t0:.0f}s")

    if not gen_structures:
        raise ValueError("No generated structures could be reconstructed.")

    # ---- Novelty ----
    print("\n--- 1. Novelty (vs MP-20 train) ---")
    t0 = time.time()
    novel_mask = compute_novelty(gen_structures, train_structures,
                                 train_formula_idx, matcher)
    n_novel = novel_mask.sum()
    pct = 100 * n_novel / len(gen_structures)
    print(f"Novel: {n_novel}/{len(gen_structures)} ({pct:.2f}%)")
    print(f"Time: {time.time() - t0:.0f}s")

    # ---- Uniqueness ----
    print("\n--- 2. Uniqueness (within generated set) ---")
    t0 = time.time()
    # Cluster ALL generated structures, count classes containing at least 1 novel
    eq_class, n_unique = compute_uniqueness(gen_structures, matcher)
    # Count one representative per unique class, and only if that
    # representative is novel. This avoids labeling a mixed class as novel.
    seen_classes = set()
    novel_unique_mask = np.zeros(len(gen_structures), dtype=bool)
    for i in range(len(gen_structures)):
        cls = eq_class[i]
        if cls in seen_classes:
            continue
        seen_classes.add(cls)
        if novel_mask[i]:
            novel_unique_mask[i] = True

    n_novel_unique = int(novel_unique_mask.sum())
    pct = 100 * n_novel_unique / len(gen_structures)
    print(f"Novel + Unique: {n_novel_unique}/{len(gen_structures)} ({pct:.2f}%)")
    print(f"  (= {n_novel_unique} novel representatives after deduplication)")
    print(f"Time: {time.time() - t0:.0f}s")

    # ---- Stability (optional) ----
    full_sun_available = False
    if not args.skip_stability:
        print("\n--- 3. Stability (CHGNet) ---")
        novel_unique_structures = [s for s, m in zip(gen_structures, novel_unique_mask) if m]
        n_to_relax = len(novel_unique_structures)

        if args.stability_n and args.stability_n < n_to_relax:
            rng = np.random.RandomState(42)
            idx = rng.choice(n_to_relax, args.stability_n, replace=False)
            relax_set = [novel_unique_structures[i] for i in idx]
            print(f"Relaxing {args.stability_n}/{n_to_relax} novel+unique structures")
        else:
            relax_set = novel_unique_structures
            print(f"Relaxing all {n_to_relax} novel+unique structures")

        t0 = time.time()
        energies_comps = compute_stability_chgnet(relax_set, device)
        relax_time = time.time() - t0
        sec_per_structure = relax_time / len(relax_set) if relax_set else 0
        print(f"Relaxation: {relax_time:.0f}s ({sec_per_structure:.1f}s/structure)")

        t0 = time.time()
        mp_api_key = os.environ.get(args.mp_api_key_env) if args.mp_api_key_env else None
        e_hull_results = compute_e_hull_batch(
            energies_comps,
            args.train_csv,
            mp_api_key=mp_api_key,
            mp_cache_path=args.mp_cache_path,
        )
        e_hull_time = time.time() - t0
        print(f"E_hull computation: {e_hull_time:.0f}s")

        n_stable = 0
        n_evaluated = 0
        n_unknown = 0
        for energy, comp in energies_comps:
            if energy is not None:
                e_hull = e_hull_results.get((energy, comp.reduced_formula))
                if e_hull is None:
                    n_unknown += 1
                else:
                    n_evaluated += 1
                    if e_hull <= args.stable_threshold:
                        n_stable += 1

        stable_pct = 100 * n_stable / n_evaluated if n_evaluated else 0
        is_sampled_stability = bool(args.stability_n and args.stability_n < n_to_relax)
        if is_sampled_stability:
            sun_pct = 100 * (n_stable / n_evaluated) * (n_novel_unique / len(gen_structures)) if n_evaluated else 0
        else:
            sun_pct = 100 * n_stable / len(gen_structures)
        full_sun_available = True
        print(
            f"Stable (E_hull <= {args.stable_threshold:g}): "
            f"{n_stable}/{n_evaluated} evaluated ({stable_pct:.2f}%)"
        )
        print(f"E_hull evaluated: {n_evaluated}/{len(relax_set)}")
        print(f"E_hull unknown:   {n_unknown}/{len(relax_set)}")
    else:
        n_stable = "N/A"
        sun_pct = None
        n_evaluated = 0
        n_unknown = 0

    # ---- Summary ----
    print("\n" + "=" * 60)
    print("S.U.N. Results" if full_sun_available else "SUN-lite Results")
    print("=" * 60)
    print(f"  Total generated:     {n_generated_total}")
    print(f"  Reconstructed:       {len(gen_structures)} ({100*len(gen_structures)/n_generated_total:.2f}%)")
    print(f"  Novel (vs train):    {n_novel} ({100*n_novel/len(gen_structures):.2f}%)")
    print(f"  Unique (generated):  {n_unique} ({100*n_unique/len(gen_structures):.2f}%)")
    print(f"  Novel + Unique:      {n_novel_unique} ({100*n_novel_unique/len(gen_structures):.2f}%)")
    print(f"  Stable:              {n_stable}")
    if full_sun_available:
        if is_sampled_stability:
            print(f"  Stability sample:    {len(relax_set)}/{n_novel_unique} novel+unique")
            print(f"  E_hull evaluated:    {n_evaluated}/{len(relax_set)}")
            print(f"  E_hull unknown:      {n_unknown}/{len(relax_set)}")
            print(f"  Sample stable rate:  {stable_pct:.2f}%")
            print(f"  Estimated S.U.N.:    {sun_pct:.2f}%")
            print("\nNote: stability was sampled; this is an estimated S.U.N. rate.")
        else:
            print(f"  S.U.N. rate:         {sun_pct:.2f}%")
        print(f"\n(Compare only when stability definitions match the benchmark.)")
    else:
        sun_lite = 100 * n_novel_unique / len(gen_structures)
        print(f"  SUN-lite rate:       {sun_lite:.2f}%")
        print("\nNote: stability was skipped, so this is Novel+Unique, not full S.U.N.")


if __name__ == "__main__":
    main()
